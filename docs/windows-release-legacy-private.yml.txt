name: Windows release

on:
  workflow_dispatch:
  push:
    branches:
      - main
      - sleepsync-integration
      - v5.1.1-development
    tags:
      - 'v*'

permissions:
  contents: write

concurrency:
  group: sleepmate-windows-${{ github.ref }}
  cancel-in-progress: true

jobs:
  build-windows-x64:
    runs-on: windows-latest
    timeout-minutes: 45
    steps:
      - name: Checkout complete source
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Python 3.13
        uses: actions/setup-python@v5
        with:
          python-version: '3.13'
          cache: 'pip'

      - name: Install Inno Setup 7.1.0
        shell: powershell
        run: |
          $url = 'https://github.com/jrsoftware/issrc/releases/download/is-7_1_0/innosetup-7.1.0-x64.exe'
          $exe = Join-Path $env:RUNNER_TEMP 'innosetup.exe'
          Invoke-WebRequest -Uri $url -OutFile $exe
          Start-Process -FilePath $exe -ArgumentList '/VERYSILENT','/SUPPRESSMSGBOXES','/NORESTART','/SP-' -Wait
          'C:\Program Files\Inno Setup 7' | Out-File -FilePath $env:GITHUB_PATH -Encoding utf8 -Append

      - name: Prepare optional Authenticode certificate
        shell: powershell
        env:
          PFX_BASE64: ${{ secrets.WINDOWS_CERT_PFX_BASE64 }}
          PFX_PASSWORD: ${{ secrets.WINDOWS_CERT_PASSWORD }}
        run: |
          if (-not $env:PFX_BASE64) {
            Write-Host 'No Windows signing certificate configured; producing an unsigned build.'
            exit 0
          }
          if (-not $env:PFX_PASSWORD) {
            throw 'WINDOWS_CERT_PFX_BASE64 is configured but WINDOWS_CERT_PASSWORD is missing.'
          }
          $pfx = Join-Path $env:RUNNER_TEMP 'sleepmate-signing.pfx'
          [IO.File]::WriteAllBytes($pfx, [Convert]::FromBase64String($env:PFX_BASE64))
          "SLEEPMATE_SIGN_PFX=$pfx" | Out-File -FilePath $env:GITHUB_ENV -Encoding utf8 -Append
          "SLEEPMATE_SIGN_PASSWORD=$($env:PFX_PASSWORD)" | Out-File -FilePath $env:GITHUB_ENV -Encoding utf8 -Append

      - name: Validate SleepMate integration contract
        shell: powershell
        run: |
          python -m pip install pytest
          python -m py_compile .\app.py .\cpap\sleepsync_integration.py .\cpap\sleepsync_engine_v2.py .\cpap\google_drive_integration.py .\cpap\v511_features.py .\sleepmate_main.py
          node --check .\web\app.js
          node --check .\web\app-core.js
          node --check .\web\mobile-boot-diagnostics.js
          node --check .\web\pwa-push-fix.js
          node --check .\web\sleepmate-enhancements.js
          node --check .\web\sleepmate-offline-runtime.js
          node --check .\web\service-worker-v508-base.js
          python .\scripts\verify_sleepsync_integration.py
          python -m pytest -q .\tests\test_release_version_contract.py .\tests\test_tailscale_v401.py .\tests\test_tailscale_v402.py .\tests\test_v511_contract.py

      - name: Build and package SleepMate
        shell: powershell
        run: |
          Write-Host 'Integrated SleepMate/SleepSync release build. The dedicated integration contract and installed-app smoke test are mandatory.'
          .\build\windows\build_release.ps1 -SkipTests

      - name: Smoke-test installed SleepMate
        shell: powershell
        run: |
          $ErrorActionPreference = 'Stop'
          $installDir = Join-Path $env:RUNNER_TEMP 'SleepMate-Smoke-Install'
          $stateDir = Join-Path $env:RUNNER_TEMP 'SleepMate-Smoke-State'
          $installLog = Join-Path $env:RUNNER_TEMP 'SleepMate-Smoke-Install.log'

          $manifestPath = 'release\sleepmate-update.json'
          if (-not (Test-Path $manifestPath)) { throw 'SleepMate update manifest was not produced.' }
          $manifest = Get-Content $manifestPath -Raw | ConvertFrom-Json
          $expectedVersion = ([string]$manifest.version).Trim()
          if (-not $expectedVersion) { throw 'SleepMate update manifest does not contain a version.' }

          $expectedZip = "release\SleepMate_v${expectedVersion}_windows_x64.zip"
          $expectedSetup = "release\SleepMate_Setup_v${expectedVersion}.exe"
          if (-not (Test-Path $expectedZip)) { throw "Expected portable ZIP was not produced: $expectedZip" }
          if (-not (Test-Path $expectedSetup)) { throw "Expected installer was not produced: $expectedSetup" }

          $setup = Get-Item $expectedSetup
          if (([string]$setup.VersionInfo.ProductVersion).Trim() -ne $expectedVersion) {
            throw "Installer ProductVersion mismatch: expected $expectedVersion, got $($setup.VersionInfo.ProductVersion)"
          }
          if ($env:GITHUB_REF -like 'refs/tags/v*') {
            $tagVersion = $env:GITHUB_REF.Substring('refs/tags/v'.Length).Trim()
            if ($tagVersion -ne $expectedVersion) {
              throw "Release tag version ($tagVersion) does not match packaged SleepMate version ($expectedVersion)."
            }
          }

          $installArgs = @('/VERYSILENT','/SUPPRESSMSGBOXES','/NORESTART','/SP-',("/DIR={0}" -f $installDir),("/LOG={0}" -f $installLog))
          $installer = Start-Process -FilePath $setup.FullName -ArgumentList $installArgs -Wait -PassThru
          if ($installer.ExitCode -ne 0) {
            if (Test-Path $installLog) { Get-Content $installLog | Write-Host }
            throw "SleepMate installer failed with exit code $($installer.ExitCode)."
          }

          $exe = Join-Path $installDir 'SleepMate.exe'
          if (-not (Test-Path $exe)) { throw "Installed SleepMate.exe not found at $exe" }
          $env:SLEEPMATE_STATE_DIR = $stateDir
          $port = 59895
          $backend = Start-Process -FilePath $exe -ArgumentList @('--backend','--host','127.0.0.1','--port',"$port",'--no-browser') -PassThru
          try {
            $version = $null
            for ($i = 0; $i -lt 40; $i++) {
              if ($backend.HasExited) { throw "Installed SleepMate backend exited early with code $($backend.ExitCode)." }
              try {
                $version = Invoke-RestMethod -Uri "http://127.0.0.1:$port/api/version" -TimeoutSec 2
                break
              } catch {
                Start-Sleep -Milliseconds 500
              }
            }
            if (-not $version) { throw 'Installed SleepMate backend did not become healthy in time.' }
            if ($version.app -ne 'SleepMate' -or ([string]$version.version).Trim() -ne $expectedVersion) {
              throw "Unexpected installed API identity: $($version | ConvertTo-Json -Compress)"
            }
            if (([string]$version.build.version).Trim() -ne $expectedVersion) {
              throw "Installed build metadata version mismatch: expected $expectedVersion, got $($version.build.version)"
            }

            $ssSettings = Invoke-RestMethod -Uri "http://127.0.0.1:$port/api/sleepsync/settings" -TimeoutSec 5
            if (-not $ssSettings.therapy_data_dir) { throw 'Installed SleepSync settings API did not return therapy_data_dir.' }
            $payload = @{ buffer_days = 2; stability_wait_seconds = 4 } | ConvertTo-Json -Compress
            $saved = Invoke-RestMethod -Uri "http://127.0.0.1:$port/api/sleepsync/settings" -Method Post -ContentType 'application/json' -Body $payload -TimeoutSec 5
            if (-not $saved.ok -or $saved.settings.buffer_days -ne 2 -or $saved.settings.stability_wait_seconds -ne 4) {
              throw "Installed SleepSync settings save failed: $($saved | ConvertTo-Json -Compress)"
            }
            $ssStatus = Invoke-RestMethod -Uri "http://127.0.0.1:$port/api/sleepsync/status" -TimeoutSec 5
            if ($null -eq $ssStatus.running -or -not $ssStatus.settings) {
              throw "Installed SleepSync status API is invalid: $($ssStatus | ConvertTo-Json -Compress)"
            }

            $drive = Invoke-RestMethod -Uri "http://127.0.0.1:$port/api/google-drive/status" -TimeoutSec 5
            if ($drive.available -ne $true -or $drive.connected -ne $false) {
              throw "Installed Google Drive optional integration status is invalid: $($drive | ConvertTo-Json -Compress)"
            }

            $push = Invoke-RestMethod -Uri "http://127.0.0.1:$port/api/push/status" -TimeoutSec 5
            if ($null -eq $push.available -or -not $push.public_key) {
              throw "Installed Web Push backend status is invalid: $($push | ConvertTo-Json -Compress)"
            }
            $webClient = New-Object System.Net.WebClient
            try {
              $swContent = $webClient.DownloadString("http://127.0.0.1:$port/service-worker.js")
              $enhancementContent = $webClient.DownloadString("http://127.0.0.1:$port/sleepmate-enhancements.js")
              $offlineContent = $webClient.DownloadString("http://127.0.0.1:$port/sleepmate-offline-runtime.js")
            } finally {
              $webClient.Dispose()
            }
            if ($swContent -notmatch "addEventListener\('push'" -or $swContent -notmatch 'cacheShellAsset') {
              throw 'Packaged service worker does not contain the resilient Web Push lifecycle.'
            }
            if ($swContent -notmatch '\[502,503,504\]' -or $swContent -match 'fresh\.status>=500') {
              throw 'Packaged service worker does not contain strict network-first offline fallback.'
            }
            if ($swContent -notmatch 'sleepmate-enhancements\.js' -or $swContent -notmatch 'sleepmate-offline-runtime\.js') {
              throw 'Packaged service worker does not precache the v5.1.1 offline runtime.'
            }
            if ($enhancementContent -notmatch 'Catmull-Rom -> Bezier' -or $enhancementContent -notmatch 'sleepmate-icon-v410\.webp') {
              throw 'Packaged v5.1.1 chart/share enhancement is missing.'
            }
            if ($offlineContent -notmatch 'offline read-only guard') {
              throw 'Packaged offline read-only guard is missing.'
            }
            Write-Host "Installed SleepMate smoke test OK: version=$expectedVersion, SleepSync, Google Drive, offline PWA and Web Push healthy."

            try {
              Invoke-RestMethod -Uri "http://127.0.0.1:$port/api/system/shutdown" -Method Post -ContentType 'application/json' -Body '{}' -TimeoutSec 5 | Out-Null
            } catch {}
            if (-not $backend.WaitForExit(10000)) {
              Stop-Process -Id $backend.Id -Force -ErrorAction SilentlyContinue
            }
          } finally {
            if (-not $backend.HasExited) {
              Stop-Process -Id $backend.Id -Force -ErrorAction SilentlyContinue
            }
          }

          $uninstaller = Join-Path $installDir 'unins000.exe'
          if (-not (Test-Path $uninstaller)) { throw 'SleepMate uninstaller was not created.' }
          $uninstall = Start-Process -FilePath $uninstaller -ArgumentList @('/VERYSILENT','/SUPPRESSMSGBOXES','/NORESTART') -Wait -PassThru
          if ($uninstall.ExitCode -ne 0) { throw "SleepMate uninstaller failed with exit code $($uninstall.ExitCode)." }

      - name: Upload build artifacts
        uses: actions/upload-artifact@v4
        with:
          name: SleepMate-Windows-x64
          path: |
            release/*
          if-no-files-found: error
          retention-days: 30

      - name: Publish stable release from main release commit
        if: github.ref == 'refs/heads/main' && startsWith(github.event.head_commit.message, 'Release SleepMate v')
        shell: powershell
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          $ErrorActionPreference = 'Stop'
          $manifest = Get-Content 'release\sleepmate-update.json' -Raw | ConvertFrom-Json
          $version = ([string]$manifest.version).Trim()
          $tag = "v$version"
          $remoteTag = & git ls-remote --tags origin "refs/tags/$tag"
          if ($remoteTag) { throw "Release tag already exists: $tag" }
          $zip = "release\SleepMate_v${version}_windows_x64.zip"
          $setup = "release\SleepMate_Setup_v${version}.exe"
          $args = @('release','create',$tag,$zip,$setup,'release\sleepmate-update.json','--repo',$env:GITHUB_REPOSITORY,'--target',$env:GITHUB_SHA,'--title',$tag)
          if (Test-Path 'RELEASE_NOTES.md') { $args += @('--notes-file','RELEASE_NOTES.md') } else { $args += '--generate-notes' }
          & gh @args
          if ($LASTEXITCODE -ne 0) { throw "GitHub Release creation failed for $tag" }

      - name: Publish manually tagged GitHub Release assets
        if: startsWith(github.ref, 'refs/tags/v')
        uses: softprops/action-gh-release@v2
        with:
          files: |
            release/SleepMate_v*_windows_x64.zip
            release/sleepmate-update.json
            release/SleepMate_Setup_v*.exe
          generate_release_notes: true
